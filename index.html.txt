<script>
// এখানে donor add করবা। তোমার দেওয়া format এ
const donors = [
  {name: "Imran Ahmed Kawser", age: 25, group: "B+", thana: "Savar", number: "01627-000", last: "2026-01-10"},
  {name: "রাহিম মিয়া", age: 24, group: "O+", thana: "ধামরাই", number: "01711-111", last: "2026-03-15"},
  {name: "করিম সাহেব", age: 30, group: "A+", thana: "সাভার", number: "01822-222222", last: "2026-04-20"}
];

function daysBetween(d1, d2) {
  return Math.floor((d2 - d1) / (1000*60*60*24));
}

function showDonors() {
  const tbody = document.querySelector("#donorTable tbody");
  const today = new Date();
  const nameVal = document.getElementById("searchName").value.toLowerCase();
  const groupVal = document.getElementById("searchGroup").value;
  const thanaVal = document.getElementById("searchThana").value;
  
  let html = "";
  
  donors.filter(d => {
    return d.name.toLowerCase().includes(nameVal) &&
           (groupVal === "" || d.group === groupVal) &&
           (thanaVal === "" || d.thana === thanaVal);
  }).forEach(d => {
    const lastDate = new Date(d.last);
    const diff = daysBetween(lastDate, today);
    let status = "";
    if(diff < 120) status = `<span class="red">🔴 ${120-diff} দিন বাকি</span>`;
    else status = `<span class="green">🟢 দিতে পারবে</span>`;
    
    html += `<tr>
      <td>${d.name}</td>
      <td>${d.age}</td>
      <td>${d.group}</td>
      <td>${d.thana}</td>
      <td><a href="tel:${d.number}" class="call-btn">📞 ${d.number}</a></td>
      <td>${d.last}</td>
      <td>${status}</td>
    </tr>`;
  });
  
  if(html === "") html = `<tr><td colspan="7">কোনো Donor পাওয়া যায়নি</td></tr>`;
  tbody.innerHTML = html;
}

showDonors();

// Search কাজ করবে
document.getElementById("searchName").onkeyup = showDonors;
document.getElementById("searchGroup").onchange = showDonors;
document.getElementById("searchThana").onchange = showDonors;
</script>
